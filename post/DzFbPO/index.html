防红|网站防qq微信举报

2026-06-17T15:16:14+08:00 · 听得入迷空间 · 技术教程

方法很简单，就是不让访客在qq微信中访问，这样就不能轻易被举报了。

以下代码放到网站里，最好是<head>标签里

<!--防qq微信屏蔽-->
<script>
// 跳转提示
if (is_weixn_qq()) {;
window.location.href = 'https://c.pc.qq.com/middle.html?pfurl='+window.location.href; 
}
// 判断QQUA的代码无需修改
function is_weixn_qq(){
var ua = navigator.userAgent.toLowerCase();
if(ua.match(/MicroMessenger/i)=="micromessenger") {
return "weixin";
} else if (ua.match(/QQ/i) == "qq") {
return "QQ";
}
return false;
}
</script>

BZQLL看到了我在网上搜到的屏蔽qq微信访问的几行js代码，还改进了。下面的代码屏蔽了qq微信，但是还可以qq浏览器访问。

<script>
    // 跳转提示
    if (is_weixn_qq()) {;
        window.location.href = 'https://c.pc.qq.com/middle.html?pfurl='+window.location.href; 
    }
    function is_weixn_qq(){
        // 判断当前是否微信/QQ浏览器
        var ua = navigator.userAgent;
        var isWeixin = !!/MicroMessenger/i.test(ua);
        var isQQ = !!/QQ\//i.test(ua);
        if(isWeixin||isQQ){
            return true;
        }
        return false;
    }
</script>
